#=============================================================================
# CMake configuration file for the Chrono::TDPF FSI module
#
# Cannot be used stand-alone (it is loaded by parent CMake configuration file)
#=============================================================================

cmake_dependent_option(CH_ENABLE_MODULE_FSI_TDPF "Enable the Chrono::TDPF FSI module" ON
                       "CH_ENABLE_MODULE_FSI" ON)

if(NOT CH_ENABLE_MODULE_FSI_TDPF)
  return()
endif()

message(STATUS "TDPF-based Chrono::FSI submodule")

# Return now if Eigen version < 3.3.6
if(EIGEN3_VERSION VERSION_LESS "3.3.6")
    message(WARNING "Eigen version (${EIGEN3_VERSION}) is less than the required version (3.3.6); disabling Chrono::FSI-TDPF")
    set(CH_ENABLE_MODULE_FSI_TDPF OFF CACHE BOOL "Enable the Chrono::TDPF FSI module" FORCE)
    return()
endif()

# Return now if no HDF5 support
if(NOT HDF5_FOUND)
    message(WARNING "Required HDF5 support not available; disabling Chrono::FSI-TDPF")
    set(CH_ENABLE_MODULE_FSI_TDPF OFF CACHE BOOL "Enable the Chrono::TDPF FSI module" FORCE)
    return()
endif()

#-----------------------------------------------------------------------------
# SEA-Stack dependency
#
# By default, use the SEA-Stack git submodule in src/chrono_thirdparty. Set
# CH_SEASTACK_DIR to the top-level directory of a different SEA-Stack tree to
# build against a local copy instead.
#
# Only the four Chrono-free library subdirectories are added, not the SEA-Stack
# top-level CMakeLists: the latter calls project(), sets policies, wires up
# CPack/install/test rules and, with SEASTACK_ENABLE_CHRONO=ON, calls
# find_package(Chrono), which would pull a second, unrelated Chrono into this
# configuration.
#-----------------------------------------------------------------------------

set(CH_SEASTACK_DIR "${CMAKE_SOURCE_DIR}/src/chrono_thirdparty/SEA-Stack" CACHE PATH "Path to top-level SEA-Stack directory")

message(STATUS "  SEA-Stack directory: ${CH_SEASTACK_DIR}")

if(NOT EXISTS "${CH_SEASTACK_DIR}/libs/hydro/CMakeLists.txt")
    if(CH_SEASTACK_DIR STREQUAL "${CMAKE_SOURCE_DIR}/src/chrono_thirdparty/SEA-Stack")
        message(WARNING "SEA-Stack submodule not populated. Run:\n"
                        "    git submodule update --init --recursive src/chrono_thirdparty/SEA-Stack\n"
                        "or set CH_SEASTACK_DIR to a local SEA-Stack tree. Disabling Chrono::FSI-TDPF")
    else()
        message(WARNING "No SEA-Stack tree found at CH_SEASTACK_DIR='${CH_SEASTACK_DIR}'; disabling Chrono::FSI-TDPF")
    endif()
    set(CH_ENABLE_MODULE_FSI_TDPF OFF CACHE BOOL "Enable the Chrono::TDPF FSI module" FORCE)
    return()
endif()

# SEA-Stack's infrastructure library includes <seastack/version.h>, normally
# generated by the SEA-Stack top-level CMakeLists.  Generate an equivalent here
# so that the library can be built stand-alone.
file(WRITE "${PROJECT_BINARY_DIR}/seastack/version.h"
"#pragma once\n"
"\n"
"#define SEASTACK_NAME         \"SEAStack\"\n"
"#define SEASTACK_VERSION      \"1.0.0\"\n"
"#define SEASTACK_VERSION_FULL \"1.0.0-beta.4\"\n")

add_subdirectory(${CH_SEASTACK_DIR}/libs/core           ${PROJECT_BINARY_DIR}/seastack/core)
add_subdirectory(${CH_SEASTACK_DIR}/libs/infrastructure ${PROJECT_BINARY_DIR}/seastack/infrastructure)
add_subdirectory(${CH_SEASTACK_DIR}/libs/hydro          ${PROJECT_BINARY_DIR}/seastack/hydro)
add_subdirectory(${CH_SEASTACK_DIR}/libs/hydro_io       ${PROJECT_BINARY_DIR}/seastack/hydro_io)

# ---------------------------------------------------------------------------
# Align the SEA-Stack libraries with Chrono's compile settings.
#
# IMPORTANT: Chrono_core applies ${SIMD_FLAGS} (/arch:AVX2 on MSVC) as a PUBLIC
# compile option, so everything linking Chrono - including Chrono_fsitdpf -
# is compiled with AVX2.  AVX2 changes Eigen's alignment from 16 to 32 bytes,
# which switches Eigen's allocator from malloc/free to
# handmade_aligned_malloc/free.  The SEA-Stack libraries do not link Chrono and
# would otherwise be compiled without AVX2; Eigen objects allocated in one and
# freed in the other would then cross allocators and corrupt the heap.
# The MSVC runtime library must match Chrono's for the same reason.
# ---------------------------------------------------------------------------

set(SEASTACK_TARGETS seastack_core seastack_infra seastack_hydro seastack_hydro_io)

foreach(ss_target ${SEASTACK_TARGETS})
    if(NOT TARGET ${ss_target})
        continue()
    endif()

    get_target_property(ss_type ${ss_target} TYPE)
    if(ss_type STREQUAL "INTERFACE_LIBRARY")
        continue()
    endif()

    target_compile_options(${ss_target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:${SIMD_FLAGS}>)
    target_include_directories(${ss_target} PRIVATE "${PROJECT_BINARY_DIR}")

    set_target_properties(${ss_target} PROPERTIES FOLDER seastack)

    if(CH_STATIC)
        set_target_properties(${ss_target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    endif()

    if(MSVC)
        set_target_properties(${ss_target} PROPERTIES MSVC_RUNTIME_LIBRARY ${CH_MSVC_RUNTIME_LIBRARY})
        target_compile_definitions(${ss_target} PRIVATE
            _CRT_SECURE_NO_DEPRECATE
            _SCL_SECURE_NO_DEPRECATE
            NOMINMAX
        )
        if(MSVC_VERSION GREATER_EQUAL 1915)
            target_compile_definitions(${ss_target} PRIVATE ENABLE_EXTENDED_ALIGNED_STORAGE)
        endif()
        target_compile_options(${ss_target} PRIVATE /wd4275 /wd4251)
    endif()
endforeach()

#-----------------------------------------------------------------------------
# List all files for the TDPF-based Chrono::FSI library
#-----------------------------------------------------------------------------

# --------------- TDPF FSI FILES

set(FSITDPF_BASE_FILES
    ChFsiTdpfTypes.h
    ChFsiSystemTDPF.h
    ChFsiSystemTDPF.cpp
    ChFsiInterfaceTDPF.h
    ChFsiInterfaceTDPF.cpp
    ChFsiFluidSystemTDPF.h
    ChFsiFluidSystemTDPF.cpp
)
source_group("" FILES ${FSITDPF_BASE_FILES})

set(FSITDPF_IMPL_FILES
    impl/ChFsiFluidSystemTDPF_impl.h
    impl/ChFsiFluidSystemTDPF_impl.cpp
)
source_group("impl" FILES ${FSITDPF_IMPL_FILES})

# --------------- TDPF FSI VSG INTERFACE

if(CH_ENABLE_MODULE_VSG)
    set(FSITDPF_VSG_FILES
        visualization/ChTdpfVisualizationVSG.h
        visualization/ChTdpfVisualizationVSG.cpp
    )
else()
    set(FSITDPF_VSG_FILES "")
endif()
source_group("" FILES ${FSITDPF_VSG_FILES})

# ------------------------------------------------------------------------------
# Set dependencies on other Chrono modules
# ------------------------------------------------------------------------------

set(DEPENDENCIES_FSI_TDPF "FSI")
if(CH_ENABLE_MODULE_VSG)
  list (APPEND DEPENDENCIES_FSI_TDPF "VSG")
endif()

set(DEPENDENCIES_FSI_TDPF ${DEPENDENCIES_FSI_TDPF} PARENT_SCOPE)

#-----------------------------------------------------------------------------
# Generate and install configuration file
#-----------------------------------------------------------------------------

set(FSITDPF_CONFIG_FILE ${PROJECT_BINARY_DIR}/chrono_fsi/tdpf/ChFsiConfigTDPF.h)

configure_file(${CMAKE_CURRENT_SOURCE_DIR}/ChFsiConfigTDPF.h.in ${FSITDPF_CONFIG_FILE} @ONLY)

source_group("" FILES ${FSITDPF_CONFIG_FILE})

install(FILES ${FSITDPF_CONFIG_FILE} DESTINATION include/chrono_fsi/tdpf)

#-----------------------------------------------------------------------------
# Create the Chrono_fsitdpf library
#-----------------------------------------------------------------------------

message(STATUS "  Add Chrono_fsitdpf library")

add_library(Chrono_fsitdpf
    ${FSITDPF_CONFIG_FILE}
    ${FSITDPF_BASE_FILES}
    ${FSITDPF_IMPL_FILES}
)
add_library(Chrono::fsitdpf ALIAS Chrono_fsitdpf)

set_target_properties(Chrono_fsitdpf PROPERTIES DEBUG_POSTFIX ${CH_DEBUG_POSTFIX})

if (CH_STATIC)
  set_target_properties(Chrono_fsitdpf PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()


if(MSVC)
  target_compile_options(Chrono_fsitdpf PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/wd4251>) # missing DLL interface
  target_compile_options(Chrono_fsitdpf PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/wd4100>) # unreferenced formal parameter

  set_target_properties(Chrono_fsitdpf PROPERTIES MSVC_RUNTIME_LIBRARY ${CH_MSVC_RUNTIME_LIBRARY})
endif()

target_link_libraries(Chrono_fsitdpf PRIVATE Chrono_core)
target_link_libraries(Chrono_fsitdpf PUBLIC Chrono_fsi)

# SEA-Stack is an implementation detail of this module: no public Chrono header
# includes a SEA-Stack header, so these stay PRIVATE.
target_link_libraries(Chrono_fsitdpf PRIVATE SEAStack::Hydro SEAStack::HydroIO)

target_compile_definitions(Chrono_fsitdpf PRIVATE $<$<COMPILE_LANGUAGE:CXX>:CH_API_COMPILE_FSI>)

install(TARGETS Chrono_fsitdpf
        EXPORT ChronoTargets
        RUNTIME DESTINATION bin
        LIBRARY DESTINATION lib
        ARCHIVE DESTINATION lib
        INCLUDES DESTINATION include/chrono_fsi)
        
#-------------------------------------------------------------------------------
# Add the Chrono_fsitdpf_vsg library
#-------------------------------------------------------------------------------

if(CH_ENABLE_MODULE_VSG)
    message(STATUS "  Add Chrono_fsitdpf_vsg library")

    add_library(Chrono_fsitdpf_vsg
        ${FSITDPF_VSG_FILES}
    )
    add_library(Chrono::fsitdpf_vsg ALIAS Chrono_fsitdpf_vsg)

    set_target_properties(Chrono_fsitdpf_vsg PROPERTIES DEBUG_POSTFIX ${CH_DEBUG_POSTFIX})

    if (CH_STATIC)
      set_target_properties(Chrono_fsitdpf_vsg PROPERTIES POSITION_INDEPENDENT_CODE ON)
    endif()
 

    if(MSVC)
      target_compile_options(Chrono_fsitdpf_vsg PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/wd4251>)  # missing DLL interface
      set_target_properties(Chrono_fsitdpf_vsg PROPERTIES MSVC_RUNTIME_LIBRARY ${CH_MSVC_RUNTIME_LIBRARY})
    endif()

    target_compile_definitions(Chrono_fsitdpf_vsg PRIVATE $<$<COMPILE_LANGUAGE:CXX>:CH_API_COMPILE_FSI>)
    
    target_link_libraries(Chrono_fsitdpf_vsg PRIVATE Chrono_core Chrono_fsitdpf)
    target_link_libraries(Chrono_fsitdpf_vsg PUBLIC Chrono_vsg)

    install(TARGETS Chrono_fsitdpf_vsg
            EXPORT ChronoTargets
            RUNTIME DESTINATION bin
            LIBRARY DESTINATION lib
            ARCHIVE DESTINATION lib
            INCLUDES DESTINATION include/chrono_fsi)

endif()
